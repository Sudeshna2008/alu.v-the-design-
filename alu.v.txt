// 4-bit ALU
// Inputs : a, b (4 bits each), op (3 bits, selects the operation)
// Outputs: result (4 bits), carry (carry out, or borrow for subtract), zero

module alu_4bit (
    input      [3:0] a,
    input      [3:0] b,
    input      [2:0] op,
    output reg [3:0] result,
    output reg       carry,
    output           zero
);

    always @(*) begin
        // Default values so the circuit never holds an old value
        result = 4'b0000;
        carry  = 1'b0;

        case (op)
            3'b000: {carry, result} = {1'b0, a} + {1'b0, b};  // ADD
            3'b001: {carry, result} = {1'b0, a} - {1'b0, b};  // SUB (carry = borrow)
            3'b010: result = a & b;                           // AND
            3'b011: result = a | b;                           // OR
            3'b100: result = a ^ b;                           // XOR
            3'b101: result = ~a;                              // NOT A
            3'b110: {carry, result} = {a, 1'b0};              // Shift A left by 1
            3'b111: {result, carry} = {1'b0, a};              // Shift A right by 1
            default: result = 4'b0000;
        endcase
    end

    // zero flag is 1 when the result is all zeros
    assign zero = (result == 4'b0000);

endmodule
